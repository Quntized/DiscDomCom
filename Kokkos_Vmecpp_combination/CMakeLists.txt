cmake_minimum_required(VERSION 3.20)
project(PlasmaGeneratorLib LANGUAGES CXX)
find_package(Kokkos REQUIRED)
find_package(KokkosFFT REQUIRED)
add_library(plasma_lib plasma_lib.cpp)
target_link_libraries(plasma_lib PUBLIC Kokkos::kokkos KokkosFFT::fft)
target_include_directories(plasma_lib PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(lib main.cpp)
target_link_libraries(lib PRIVATE plasma_lib)
